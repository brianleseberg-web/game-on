THE PLAYERS ---------- // This function builds one player. We use it twice, once for each player. function makePlayer(name, label, color, controls) { return { name: name, label: label, // the short name drawn above the player color: color, controls: controls, // which keys move this player x: 0, // position across the screen y: 0, // position down the screen width: 28, height: 28, speed: 3.5, // how many pixels we move each frame score: 0, facing: 1, // 1 = looking right, -1 = looking left hitTimer: 0 // counts down after getting hit (makes the flashing) }; } // Player 1 is blue and uses W A S D const player1 = makePlayer("Player 1", "P1", "#2f6fed", { up: "KeyW", down: "KeyS", left: "KeyA", right: "KeyD" }); // Player 2 is pink and uses the arrow keys const player2 = makePlayer("Player 2", "P2", "#e83e8c", { up: "ArrowUp", down: "ArrowDown", left: "ArrowLeft", right: "ArrowRight" }); // Put both players in one list so we can loop over them together const players = [player1, player2]; // ---------- 4. ---------- 5. LOADING A LEVEL ---------- // Sets up everything for one level: walls, enemies, gems and player positions function loadLevel(index) { const data = levels[index]; // Walls, spikes and the goal never change, so we can use them directly walls = data.walls; spikes = data.spikes; goal = data.goal; // Enemies move, so we make fresh copies. That way every new game // starts with the enemies back in their first positions. movers = []; for (let i = 0; i < data.movers.length; i++) { const m = data.movers[i]; movers.push({ x: m.x, y: m.y, w: m.w, h: m.h, axis: m.axis, min: m.min, max: m.max, speed: m.speed, direction: 1 }); } // Gems get collected, so we also make fresh copies (none collected yet) gems = []; for (let i = 0; i < data.gems.length; i++) { const g = data.gems[i]; gems.push({ x: g.x, y: g.y, taken: false }); } // Put both players at their starting spots and show the level name sendToStart(player1); sendToStart(player2); levelWinner = null; levelNameText.textContent = data.name; } // Moves one player back to the starting spot of the current level function sendToStart(player) { const data = levels[currentLevel]; // Player 1 and Player 2 have different starting spots let spot; if (player === player1) { spot = data.start1; } else { spot = data.start2; } player.x = spot.x; player.y = spot.y; } // ---------- 6. HELPER FUNCTIONS ---------- // Checks if two rectangles overlap. Returns true if they touch, false if not. // We use this for walls, enemies, gems and the flag. function isTouching(a, b) { return a.x < b.x + b.width && a.x + a.width > b.x && a.y < b.y + b.height && a.y + a.height > b.y; } // Our level data uses short names (w and h). This converts a level item // into a rectangle that isTouching() understands. function toRect(thing) { return { x: thing.x, y: thing.y, width: thing.w, height: thing.h }; } // Gems are stored by their centre point, so we give each one a small box function gemRect(gem) { return { x: gem.x - 10, y: gem.y - 10, width: 20, height: 20 }; } // ---------- 7. UPDATING THE GAME (runs 60 times per second) ---------- // Reads the keys a player is holding and moves them function movePlayer(player) { let moveX = 0; let moveY = 0; // Work out which way the player wants to go if (keys[player.controls.left]) { moveX = -1; player.facing = -1; } if (keys[player.controls.right]) { moveX = 1; player.facing = 1; } if (keys[player.controls.up]) { moveY = -1; } if (keys[player.controls.down]) { moveY = 1; } // Moving diagonally would be faster, so we slow it down a little if (moveX !== 0 && moveY !== 0) { moveX = moveX * 0.7; moveY = moveY * 0.7; } // Move sideways first, then up/down. Doing one at a time lets // players slide along walls instead of getting stuck. stepPlayer(player, moveX * player.speed, 0); stepPlayer(player, 0, moveY * player.speed); } // Moves a player one small step, and undoes the step if they hit a wall function stepPlayer(player, stepX, stepY) { // Remember where we were, in case we need to go back const oldX = player.x; const oldY = player.y; // Take the step player.x = player.x + stepX; player.y = player.y + stepY; // Keep the player inside the game area if (player.x < 0) player.x = 0; if (player.y < 0) player.y = 0; if (player.x > WIDTH - player.width) player.x = WIDTH - player.width; if (player.y > HEIGHT - player.height) player.y = HEIGHT - player.height; // If we are now inside a wall, go back to where we were for (let i = 0; i < walls.length; i++) { if (isTouching(player, toRect(walls[i]))) { player.x = oldX; player.y = oldY; return: } } } // Slides every enemy back and forth between its min and max positions function moveEnemies() { for (let i = 0; i < movers.length; i++) { const m = movers[i]; if (m.axis === "x") { // Move left/right, and turn around at either end m.x = m.x + m.speed * m.direction; if (m.x > m.max) m.direction = -1; if (m.x < m.min) m.direction = 1; } else { // Move up/down, and turn around at either end m.y = m.y + m.speed * m.direction; if (m.y > m.max) m.direction = -1; if (m.y < m.min) m.direction = 1; } } } // Sends a player back to the start and makes them flash for a moment function hurtPlayer(player) { sendToStart(player); player.hitTimer = 40; } // Checks if a player touched an enemy or a spike function checkDanger(player) { // Check every enemy for (let i = 0; i < movers.length; i++) { if (isTouching(player, toRect(movers[i]))) { hurtPlayer(player); return; } } // Check every spike for (let i = 0; i < spikes.length; i++) { if (isTouching(player, toRect(spikes[i]))) { hurtPlayer(player); return; } } } // Checks if a player touched any gems. Collected gems are worth 1 point. function checkGems(player) { for (let i = 0; i < gems.length; i++) { const gem = gems[i]; if (gem.taken === false && isTouching(player, gemRect(gem))) { gem.taken = true; player.score = player.score + 1; } } } // Checks if a player reached the flag. The first one there wins 5 points // and the level ends. function checkGoal(player) { if (levelWinner === null && isTouching(player, toRect(goal))) { levelWinner = player; player.score = player.score + 5; state = "levelOver"; } } // Does one frame of game logic. Nothing moves unless we are playing. function update() { if (state !== "playing") return; moveEnemies(); // Run the same checks for both players for (let i = 0; i < players.length; i++) { const player = players[i]; movePlayer(player); checkDanger(player); checkGems(player); checkGoal(player); // Count down the flashing timer after a hit if (player.hitTimer > 0) { player.hitTimer = player.hitTimer - 1; } } updateScoreboard(); } // Copies both players' scores onto the page function updateScoreboard() { scoreP1Text.textContent = player1.score; scoreP2Text.textContent = player2.score; } // ---------- 8.
