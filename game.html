// GEM RUSH - a simple 2 player game // Player 1: W A S D Player 2: Arrow keys // Collect gems, avoid enemies and spikes, reach the flag first! // ---------- 1. SETUP ---------- // Find the canvas on the page, and get the "pen" we use to draw on it const canvas = document.getElementById("gameCanvas"); const ctx = canvas.getContext("2d"); // Remember the size of the game area so we can use it later const WIDTH = canvas.width; const HEIGHT = canvas.height; // Find the pieces of text on the page that we change while playing const scoreP1Text = document.getElementById("scoreP1"); const scoreP2Text = document.getElementById("scoreP2"); const levelNameText = document.getElementById("levelName"); // ---------- 2. GAME STATE ---------- // These variables remember what is happening in the game right now. // The game is always in one of four states: // "menu", "playing", "levelOver" or "gameOver" let state = "menu"; // Which level we are on (0 = level 1, 1 = level 2, 2 = level 3) let currentLevel = 0; // Who reached the flag first on this level (empty until someone does) let levelWinner = null; // These lists are filled in by loadLevel() for whichever level is active let walls = []; let spikes = []; let movers = []; let gems = []; let goal = null; // Remembers which keys are being held down right now const keys = {}; // ---------- 3. THE PLAYERS ---------- // This function builds one player. We use it twice, once for each player. function makePlayer(name, label, color, controls) { return { name: name, label: label, // the short name drawn above the player color: color, controls: controls, // which keys move this player x: 0, // position across the screen y: 0, // position down the screen width: 28, height: 28, speed: 3.5, // how many pixels we move each frame score: 0, facing: 1, // 1 = looking right, -1 = looking left hitTimer: 0 // counts down after getting hit (makes the flashing) }; } // Player 1 is blue and uses W A S D const player1 = makePlayer("Player 1", "P1", "#2f6fed", { up: "KeyW", down: "KeyS", left: "KeyA", right: "KeyD" }); // Player 2 is pink and uses the arrow keys const player2 = makePlayer("Player 2", "P2", "#e83e8c", { up: "ArrowUp", down: "ArrowDown", left: "ArrowLeft", right: "ArrowRight" }); // Put both players in one list so we can loop over them together const players = [player1, player2]; // ---------- 4. THE LEVELS ---------- // Each level is a list of rectangles written as {x, y, w, h}. // Change the numbers to design your own levels! // Movers slide back and forth: axis "x" = left/right, axis "y" = up/down. const levels = [ // LEVEL 1: Green Meadow (the easiest level) { name: "Level 1: Green Meadow", theme: "grass", // picks the background wallColor: "#8b5a2b", start1: { x: 30, y: 60 }, // where Player 1 starts start2: { x: 30, y: 410 }, // where Player 2 starts goal: { x: 730, y: 220, w: 40, h: 60 }, // the flag walls: [ { x: 200, y: 0, w: 30, h: 200 }, { x: 200, y: 300, w: 30, h: 200 }, { x: 450, y: 100, w: 30, h: 300 } ], movers: [ { x: 330, y: 20, w: 30, h: 30, axis: "y", min: 20, max: 450, speed: 2 }, { x: 580, y: 450, w: 30, h: 30, axis: "y", min: 20, max: 450, speed: 3 } ], spikes: [], // no spikes on level 1 gems: [ { x: 120, y: 250 }, { x: 380, y: 60 }, { x: 380, y: 440 }, { x: 520, y: 250 }, { x: 650, y: 100 }, { x: 650, y: 400 } ] }, // LEVEL 2: Hot Desert (faster enemies and some spikes) { name: "Level 2: Hot Desert", theme: "desert", wallColor: "#a0522d", start1: { x: 30, y: 40 }, start2: { x: 30, y: 430 }, goal: { x: 730, y: 400, w: 40, h: 60 }, walls: [ { x: 120, y: 120, w: 200, h: 30 }, { x: 120, y: 350, w: 200, h: 30 }, { x: 400, y: 0, w: 30, h: 220 }, { x: 400, y: 280, w: 30, h: 220 }, { x: 600, y: 150, w: 30, h: 200 } ], movers: [ { x: 150, y: 235, w: 30, h: 30, axis: "x", min: 60, max: 300, speed: 3 }, { x: 500, y: 20, w: 30, h: 30, axis: "y", min: 20, max: 450, speed: 3 }, { x: 450, y: 60, w: 30, h: 30, axis: "x", min: 440, max: 720, speed: 3 }, { x: 680, y: 440, w: 30, h: 30, axis: "x", min: 440, max: 720, speed: 2 } ], spikes: [ { x: 330, y: 235, w: 40, h: 30 }, { x: 660, y: 200, w: 40, h: 30 } ], gems: [ { x: 80, y: 250 }, { x: 220, y: 60 }, { x: 220, y: 440 }, { x: 360, y: 180 }, { x: 360, y: 320 }, { x: 560, y: 60 }, { x: 680, y: 300 } ] }, // LEVEL 3: Frozen Night (the hardest level, with the most hazards) { name: "Level 3: Frozen Night", theme: "ice", wallColor: "#5b7fa6", start1: { x: 30, y: 60 }, start2: { x: 30, y: 410 }, goal: { x: 730, y: 220, w: 40, h: 60 }, walls: [ { x: 100, y: 150, w: 30, h: 200 }, { x: 250, y: 0, w: 30, h: 180 }, { x: 250, y: 320, w: 30, h: 180 }, { x: 400, y: 130, w: 30, h: 240 }, { x: 550, y: 0, w: 30, h: 200 }, { x: 550, y: 300, w: 30, h: 200 } ], movers: [ { x: 170, y: 20, w: 30, h: 30, axis: "y", min: 20, max: 450, speed: 3 }, { x: 330, y: 450, w: 30, h: 30, axis: "y", min: 20, max: 450, speed: 4 }, { x: 430, y: 40, w: 30, h: 30, axis: "x", min: 435, max: 515, speed: 3 }, { x: 435, y: 440, w: 30, h: 30, axis: "x", min: 435, max: 515, speed: 3 }, { x: 650, y: 20, w: 30, h: 30, axis: "y", min: 20, max: 450, speed: 4 } ], spikes: [ { x: 480, y: 240, w: 40, h: 20 }, { x: 150, y: 250, w: 40, h: 20 } ], gems: [ { x: 60, y: 250 }, { x: 210, y: 100 }, { x: 340, y: 250 }, { x: 470, y: 60 }, { x: 470, y: 440 }, { x: 610, y: 250 }, { x: 700, y: 100 }, { x: 700, y: 400 } ] } ]; // ---------- 5. LOADING A LEVEL ---------- // Sets up everything for one level: walls, enemies, gems and player positions function loadLevel(index) { const data = levels[index]; // Walls, spikes and the goal never change, so we can use them directly walls = data.walls; spikes = data.spikes; goal = data.goal; // Enemies move, so we make fresh copies. That way every new game // starts with the enemies back in their first positions. movers = []; for (let i = 0; i < data.movers.length; i++) { const m = data.movers[i]; movers.push({ x: m.x, y: m.y, w: m.w, h: m.h, axis: m.axis, min: m.min, max: m.max, speed: m.speed, direction: 1 }); } // Gems get collected, so we also make fresh copies (none collected yet) gems = []; for (let i = 0; i < data.gems.length; i++) { const g = data.gems[i]; gems.push({ x: g.x, y: g.y, taken: false }); } // Put both players at their starting spots and show the level name sendToStart(player1); sendToStart(player2); levelWinner = null; levelNameText.textContent = data.name; } // Moves one player back to the starting spot of the current level function sendToStart(player) { const data = levels[currentLevel]; // Player 1 and Player 2 have different starting spots let spot; if (player === player1) { spot = data.start1; } else { spot = data.start2; } player.x = spot.x; player.y = spot.y; } // ---------- 6. HELPER FUNCTIONS ---------- // Checks if two rectangles overlap. Returns true if they touch, false if not. // We use this for walls, enemies, gems and the flag. function isTouching(a, b) { return a.x < b.x + b.width && a.x + a.width > b.x && a.y < b.y + b.height && a.y + a.height > b.y; } // Our level data uses short names (w and h). This converts a level item // into a rectangle that isTouching() understands. function toRect(thing) { return { x: thing.x, y: thing.y, width: thing.w, height: thing.h }; } // Gems are stored by their centre point, so we give each one a small box function gemRect(gem) { return { x: gem.x - 10, y: gem.y - 10, width: 20, height: 20 }; } // ---------- 7. UPDATING THE GAME (runs 60 times per second) ---------- // Reads the keys a player is holding and moves them function movePlayer(player) { let moveX = 0; let moveY = 0; // Work out which way the player wants to go if (keys[player.controls.left]) { moveX = -1; player.facing = -1; } if (keys[player.controls.right]) { moveX = 1; player.facing = 1; } if (keys[player.controls.up]) { moveY = -1; } if (keys[player.controls.down]) { moveY = 1; } // Moving diagonally would be faster, so we slow it down a little if (moveX !== 0 && moveY !== 0) { moveX = moveX * 0.7; moveY = moveY * 0.7; } // Move sideways first, then up/down. Doing one at a time lets // players slide along walls instead of getting stuck. stepPlayer(player, moveX * player.speed, 0); stepPlayer(player, 0, moveY * player.speed); } // Moves a player one small step, and undoes the step if they hit a wall function stepPlayer(player, stepX, stepY) { // Remember where we were, in case we need to go back const oldX = player.x; const oldY = player.y; // Take the step player.x = player.x + stepX; player.y = player.y + stepY; // Keep the player inside the game area if (player.x < 0) player.x = 0; if (player.y < 0) player.y = 0; if (player.x > WIDTH - player.width) player.x = WIDTH - player.width; if (player.y > HEIGHT - player.height) player.y = HEIGHT - player.height; // If we are now inside a wall, go back to where we were for (let i = 0; i < walls.length; i++) { if (isTouching(player, toRect(walls[i]))) { player.x = oldX; player.y = oldY; return: } } } // Slides every enemy back and forth between its min and max positions function moveEnemies() { for (let i = 0; i < movers.length; i++) { const m = movers[i]; if (m.axis === "x") { // Move left/right, and turn around at either end m.x = m.x + m.speed * m.direction; if (m.x > m.max) m.direction = -1; if (m.x < m.min) m.direction = 1; } else { // Move up/down, and turn around at either end m.y = m.y + m.speed * m.direction; if (m.y > m.max) m.direction = -1; if (m.y < m.min) m.direction = 1; } } } // Sends a player back to the start and makes them flash for a moment function hurtPlayer(player) { sendToStart(player); player.hitTimer = 40; } // Checks if a player touched an enemy or a spike function checkDanger(player) { // Check every enemy for (let i = 0; i < movers.length; i++) { if (isTouching(player, toRect(movers[i]))) { hurtPlayer(player); return; } } // Check every spike for (let i = 0; i < spikes.length; i++) { if (isTouching(player, toRect(spikes[i]))) { hurtPlayer(player); return; } } } // Checks if a player touched any gems. Collected gems are worth 1 point. function checkGems(player) { for (let i = 0; i < gems.length; i++) { const gem = gems[i]; if (gem.taken === false && isTouching(player, gemRect(gem))) { gem.taken = true; player.score = player.score + 1; } } } // Checks if a player reached the flag. The first one there wins 5 points // and the level ends. function checkGoal(player) { if (levelWinner === null && isTouching(player, toRect(goal))) { levelWinner = player; player.score = player.score + 5; state = "levelOver"; } } // Does one frame of game logic. Nothing moves unless we are playing. function update() { if (state !== "playing") return; moveEnemies(); // Run the same checks for both players for (let i = 0; i < players.length; i++) { const player = players[i]; movePlayer(player); checkDanger(player); checkGems(player); checkGoal(player); // Count down the flashing timer after a hit if (player.hitTimer > 0) { player.hitTimer = player.hitTimer - 1; } } updateScoreboard(); } // Copies both players' scores onto the page function updateScoreboard() { scoreP1Text.textContent = player1.score; scoreP2Text.textContent = player2.score; } // ---------- 8. DRAWING THE BACKGROUNDS ---------- // Positions for the little decorations on each background const flowers = [ { x: 60, y: 40 }, { x: 150, y: 120 }, { x: 90, y: 300 }, { x: 170, y: 460 }, { x: 270, y: 60 }, { x: 300, y: 380 }, { x: 360, y: 200 }, { x: 420, y: 470 }, { x: 520, y: 50 }, { x: 600, y: 330 }, { x: 690, y: 450 }, { x: 750, y: 80 } ]; const dunes = [ { x: 80, y: 120 }, { x: 250, y: 260 }, { x: 420, y: 400 }, { x: 590, y: 130 }, { x: 700, y: 300 }, { x: 200, y: 440 } ]; const pebbles = [ { x: 40, y: 200 }, { x: 130, y: 380 }, { x: 210, y: 90 }, { x: 300, y: 300 }, { x: 340, y: 470 }, { x: 410, y: 150 }, { x: 480, y: 340 }, { x: 540, y: 60 }, { x: 610, y: 240 }, { x: 660, y: 420 }, { x: 720, y: 130 }, { x: 770, y: 330 }, { x: 500, y: 480 }, { x: 90, y: 470 } ]; const snowflakes = [ { x: 30, y: 60 }, { x: 90, y: 210 }, { x: 140, y: 340 }, { x: 190, y: 30 }, { x: 240, y: 430 }, { x: 310, y: 120 }, { x: 360, y: 290 }, { x: 420, y: 60 }, { x: 470, y: 380 }, { x: 520, y: 200 }, { x: 580, y: 470 }, { x: 630, y: 90 }, { x: 680, y: 320 }, { x: 730, y: 200 }, { x: 770, y: 440 }, { x: 60, y: 430 }, { x: 330, y: 480 }, { x: 600, y: 20 }, { x: 450, y: 250 }, { x: 250, y: 180 } ]; // Level 1 background: checkered grass with little flowers function drawGrassBackground() { // Draw a grid of squares, swapping between dark and light green let isDark = true; for (let x = 0; x < WIDTH; x += 50) { for (let y = 0; y < HEIGHT; y += 50) { if (isDark) { ctx.fillStyle = "#5fae4e"; } else { ctx.fillStyle = "#6cc05a"; } ctx.fillRect(x, y, 50, 50); // Swap colour for the next square down isDark = !isDark; } // Swap once more at the end of each column to make a checker pattern isDark = !isDark; } // Draw each flower from the list for (let i = 0; i < flowers.length; i++) { // White petals ctx.fillStyle = "white"; ctx.beginPath(); ctx.arc(flowers[i].x, flowers[i].y, 5, 0, Math.PI * 2); ctx.fill(); // Yellow middle ctx.fillStyle = "#ffd84d"; ctx.beginPath(); ctx.arc(flowers[i].x, flowers[i].y, 2, 0, Math.PI * 2); ctx.fill(); } } // Level 2 background: sand with dunes and pebbles function drawDesertBackground() { // Flat sandy colour ctx.fillStyle = "#e8c978"; ctx.fillRect(0, 0, WIDTH, HEIGHT); // Darker oval shapes look like sand dunes ctx.fillStyle = "#dfb95f"; for (let i = 0; i < dunes.length; i++) { ctx.beginPath(); ctx.ellipse(dunes[i].x, dunes[i].y, 110, 30, 0, 0, Math.PI * 2); ctx.fill(); } // Tiny rectangles look like pebbles ctx.fillStyle = "#b8924a"; for (let i = 0; i < pebbles.length; i++) { ctx.fillRect(pebbles[i].x, pebbles[i].y, 4, 3); } } // Level 3 background: dark icy night with snow function drawIceBackground() { // A gradient fades smoothly from dark blue at the top to lighter blue below const gradient = ctx.createLinearGradient(0, 0, 0, HEIGHT); gradient.addColorStop(0, "#14234d"); gradient.addColorStop(1, "#3b6ea8"); ctx.fillStyle = gradient; ctx.fillRect(0, 0, WIDTH, HEIGHT); // Faint vertical lines make the ice look like tiles ctx.strokeStyle = "rgba(255, 255, 255, 0.08)"; for (let x = 0; x < WIDTH; x += 50) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, HEIGHT); ctx.stroke(); } // Faint horizontal lines to finish the tile pattern for (let y = 0; y < HEIGHT; y += 50) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(WIDTH, y); ctx.stroke(); } // White dots are the snowflakes ctx.fillStyle = "white"; for (let i = 0; i < snowflakes.length; i++) { ctx.beginPath(); ctx.arc(snowflakes[i].x, snowflakes[i].y, 2, 0, Math.PI * 2); ctx.fill(); } } // Picks the right background for whichever level we are on function drawBackground() { const theme = levels[currentLevel].theme; if (theme === "grass") { drawGrassBackground(); } if (theme === "desert") { drawDesertBackground(); } if (theme === "ice") { drawIceBackground(); } } // ---------- 9. DRAWING THE SPRITES ---------- // Draws the solid walls players cannot walk through function drawWalls() { const color = levels[currentLevel].wallColor; for (let i = 0; i < walls.length; i++) { const wall = walls[i]; // The main wall block ctx.fillStyle = color; ctx.fillRect(wall.x, wall.y, wall.w, wall.h); // A lighter strip on the left edge makes it look 3D ctx.fillStyle = "rgba(255, 255, 255, 0.25)"; ctx.fillRect(wall.x, wall.y, 6, wall.h); // A thin dark outline ctx.strokeStyle = "rgba(0, 0, 0, 0.5)"; ctx.strokeRect(wall.x, wall.y, wall.w, wall.h); } } // Draws the goal flag that players race to function drawGoal() { // The pole ctx.fillStyle = "#dddddd"; ctx.fillRect(goal.x + 8, goal.y, 5, goal.h); // The yellow flag (a triangle) ctx.fillStyle = "#ffd84d"; ctx.beginPath(); ctx.moveTo(goal.x + 13, goal.y); ctx.lineTo(goal.x + 40, goal.y + 14); ctx.lineTo(goal.x + 13, goal.y + 28); ctx.closePath(); ctx.fill(); // The grey base at the bottom of the pole ctx.fillStyle = "#555555"; ctx.fillRect(goal.x, goal.y + goal.h - 6, goal.w, 6); } // Draws the spikes as a row of silver triangles function drawSpikes() { for (let i = 0; i < spikes.length; i++) { const spike = spikes[i]; // One triangle for every 10 pixels of width ctx.fillStyle = "#c0c0c0"; for (let j = 0; j < spike.w; j += 10) { ctx.beginPath(); ctx.moveTo(spike.x + j, spike.y + spike.h); ctx.lineTo(spike.x + j + 5, spike.y); ctx.lineTo(spike.x + j + 10, spike.y + spike.h); ctx.closePath(); ctx.fill(); } // A dark strip along the bottom of the spikes ctx.fillStyle = "#555555"; ctx.fillRect(spike.x, spike.y + spike.h - 3, spike.w, 3); } } // Draws every gem that has not been collected yet, as a little diamond function drawGems() { for (let i = 0; i < gems.length; i++) { const gem = gems[i]; // Skip gems that were already collected if (gem.taken) { continue; } // The blue diamond shape ctx.fillStyle = "#3de0ff"; ctx.beginPath(); ctx.moveTo(gem.x, gem.y - 11); ctx.lineTo(gem.x + 9, gem.y); ctx.lineTo(gem.x, gem.y + 11); ctx.lineTo(gem.x - 9, gem.y); ctx.closePath(); ctx.fill(); // A white outline ctx.strokeStyle = "white"; ctx.lineWidth = 2; ctx.stroke(); // A tiny white square for the shine ctx.fillStyle = "white"; ctx.fillRect(gem.x - 3, gem.y - 5, 3, 3); } } // Draws the moving enemies as red squares with angry faces function drawEnemies() { for (let i = 0; i < movers.length; i++) { const m = movers[i]; // Red body with a dark red outline ctx.fillStyle = "#d93025"; ctx.fillRect(m.x, m.y, m.w, m.h); ctx.strokeStyle = "#7a0f09"; ctx.lineWidth = 2; ctx.strokeRect(m.x, m.y, m.w, m.h); // White eyes with black pupils ctx.fillStyle = "white"; ctx.fillRect(m.x + 6, m.y + 9, 6, 6); ctx.fillRect(m.x + 18, m.y + 9, 6, 6); ctx.fillStyle = "black"; ctx.fillRect(m.x + 8, m.y + 11, 3, 3); ctx.fillRect(m.x + 20, m.y + 11, 3, 3); // Angry eyebrows (two slanted lines) and a flat mouth ctx.strokeStyle = "black"; ctx.beginPath(); ctx.moveTo(m.x + 5, m.y + 6); ctx.lineTo(m.x + 13, m.y + 10); ctx.moveTo(m.x + 25, m.y + 6); ctx.lineTo(m.x + 17, m.y + 10); ctx.stroke(); ctx.fillRect(m.x + 9, m.y + 21, 12, 3); } } // Draws one player: a coloured square with eyes, a smile and a label function drawPlayer(player) { // After being hit the player turns see-through for a moment if (player.hitTimer > 0) { ctx.globalAlpha = 0.4; } // A soft shadow under the player ctx.fillStyle = "rgba(0, 0, 0, 0.3)"; ctx.beginPath(); ctx.ellipse(player.x + 14, player.y + 29, 13, 4, 0, 0, Math.PI * 2); ctx.fill(); // The coloured body with a white outline ctx.fillStyle = player.color; ctx.fillRect(player.x, player.y, player.width, player.height); ctx.strokeStyle = "white"; ctx.lineWidth = 2; ctx.strokeRect(player.x, player.y, player.width, player.height); // Eyes: the pupils shift left or right to show which way we are facing const lookX = player.facing * 2; ctx.fillStyle = "white"; ctx.fillRect(player.x + 5, player.y + 7, 7, 8); ctx.fillRect(player.x + 16, player.y + 7, 7, 8); ctx.fillStyle = "black"; ctx.fillRect(player.x + 7 + lookX, player.y + 10, 3, 4); ctx.fillRect(player.x + 18 + lookX, player.y + 10, 3, 4); // The smile ctx.fillStyle = "white"; ctx.fillRect(player.x + 9, player.y + 20, 10, 3); // "P1" or "P2" written above the player's head ctx.font = "bold 12px Arial"; ctx.textAlign = "center"; ctx.fillText(player.label, player.x + 14, player.y - 5); // Go back to normal (not see-through) for everything drawn next ctx.globalAlpha = 1; } // ---------- 10. MENUS AND MESSAGES ---------- // Draws a dark box over the game with a big title and lines of text function drawMessageBox(title, lines) { // Darken the whole screen ctx.fillStyle = "rgba(0, 0, 0, 0.75)"; ctx.fillRect(0, 0, WIDTH, HEIGHT); // The big yellow title ctx.textAlign = "center"; ctx.fillStyle = "#ffd84d"; ctx.font = "bold 44px Arial"; ctx.fillText(title, WIDTH / 2, 150); // The smaller white lines, one under the other ctx.fillStyle = "white"; ctx.font = "22px Arial"; for (let i = 0; i < lines.length; i++) { ctx.fillText(lines[i], WIDTH / 2, 215 + i * 36); } } // The title screen shown before the game starts function drawMenu() { drawMessageBox("GEM RUSH", [ "Two players. Three levels. One winner.", "Player 1: W A S D Player 2: Arrow keys", "Collect gems (1 point). Reach the flag first (5 points).", "Avoid the red enemies and the spikes!", "", "Press ENTER to start" ]); } // The message shown when someone reaches the flag function drawLevelOver() { // On the last level the button leads to the results instead of a new level let nextMessage = "Press ENTER for the next level"; if (currentLevel === levels.length - 1) { nextMessage = "Press ENTER to see the results"; } drawMessageBox(levelWinner.name + " reached the flag!", [ "+5 points for " + levelWinner.name, "", "Player 1: " + player1.score + " Player 2: " + player2.score, "", nextMessage ]); } // The final screen: shows who won after all three levels function drawGameOver() { // Start by assuming a tie, then change it if one player is ahead let title = "It's a tie!"; if (player1.score > player2.score) { title = "Player 1 wins!"; } if (player2.score > player1.score) { title = "Player 2 wins!"; } drawMessageBox(title, [ "Final score", "", "Player 1: " + player1.score + " Player 2: " + player2.score, "", "Press ENTER to play again" ]); } // ---------- 11. THE MAIN DRAW FUNCTION ---------- // Draws everything for one frame, from back to front like layers of paper function draw() { // The background goes first, at the very back drawBackground(); // On the menu we only need the background and the menu box if (state === "menu") { drawMenu(); return; } // Then the level items, then the players on top drawWalls(); drawGoal(); drawSpikes(); drawGems(); drawEnemies(); drawPlayer(player1); drawPlayer(player2); // Messages go on the very top of everything if (state === "levelOver") { drawLevelOver(); } if (state === "gameOver") { drawGameOver(); } } // ---------- 12. KEYBOARD INPUT ---------- // Runs whenever a key is pressed down function handleKeyDown(event) { // Remember that this key is being held keys[event.code] = true; // Stop the arrow keys from scrolling the web page up and down if (event.code === "ArrowUp" || event.code === "ArrowDown" || event.code === "ArrowLeft" || event.code === "ArrowRight") { event.preventDefault(); } // Enter is used to move through the menus if (event.code === "Enter") { handleEnterKey(); } } // Runs whenever a key is let go function handleKeyUp(event) { // Remember that this key is no longer held keys[event.code] = false; } // Tell the browser to run the two functions above when keys are pressed and released window.addEventListener("keydown", handleKeyDown); window.addEventListener("keyup", handleKeyUp); // Decides what the Enter key does, depending on the state of the game function handleEnterKey() { if (state === "menu" || state === "gameOver") { startNewGame(); } else if (state === "levelOver") { goToNextLevel(); } } // ---------- 13. STARTING AND FINISHING LEVELS ---------- // Resets the scores and begins again from level 1 function startNewGame() { player1.score = 0; player2.score = 0; currentLevel = 0; loadLevel(currentLevel); updateScoreboard(); state = "playing"; } // Moves on to the next level, or shows the final results after the last one function goToNextLevel() { currentLevel = currentLevel + 1; if (currentLevel >= levels.length) { // No more levels: stay on the last one and show the results currentLevel = levels.length - 1; state = "gameOver"; } else { loadLevel(currentLevel); state = "playing"; } } // ---------- 14. THE GAME LOOP ---------- // One tick of the game: update the logic, then draw the picture function gameLoop() { update(); draw(); } // Load level 1 first so the menu screen has a background to show loadLevel(0); // Run the game loop 60 times every second setInterval(gameLoop, 1000 / 60);
